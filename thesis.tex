% Josephson-junction pics: jj (no junction capacitance) and jjbox (with capacitance).
\tikzset{
  % Boxed Josephson junction
  pics/jjbox/.style={
    code={
      % leads
      \draw[line width=1.1pt] (-0.65,0) -- (-0.25,0);
      \draw[line width=1.1pt] ( 0.25,0) -- ( 0.65,0);

      % box
      \draw[line width=1.1pt]
        (-0.25,-0.25) rectangle (0.25,0.25);

      % X -- extends to the box corners
      \draw[line width=1.1pt]
        (-0.25,-0.25) -- (0.25,0.25);
      \draw[line width=1.1pt]
        (-0.25, 0.25) -- (0.25,-0.25);
    }
  },
  % Unboxed Josephson junction
  pics/jj/.style={
    code={
      % leads
      \draw[line width=1.1pt] (-0.65,0) -- (0,0);
      \draw[line width=1.1pt] ( 0.0,0) -- ( 0.65,0);

      % X -- extends to the box corners
      \draw[line width=1.1pt]
        (-0.25,-0.25) -- (0.25,0.25);
      \draw[line width=1.1pt]
        (-0.25, 0.25) -- (0.25,-0.25);
    }
  }
}
