% Single-mode qubit landscape: columns add an inductor (E_L = 0 vs. E_L > 0),
% rows add capacitance (small vs. large E_J/E_C).
\tikzset{
  bigC/.style={C, capacitors/scale=1.7},
  smallC/.style={C, capacitors/scale=0.7},
  desc/.style={text width=3.6cm, align=center, font=\small, anchor=north},
  axis/.style={-latex, line width=0.8pt, gray!70!black},
}

% Shunted junction: capacitor style #1 at (#2), width 1.8 and height 1.6.
\newcommand{\cpbcircuit}[2]{%
  \draw (#2) ++(-0.9,0) coordinate (bl) ++(0,1.6) coordinate (tl)
        (#2) ++(0.9,0) coordinate (br) ++(0,1.6) coordinate (tr);
  \draw (bl) to[#1, l=$C$] (tl) -- (tr) to[josephson, l=$E_J$] (br) -- (bl);
}
% Fluxonium: C, L and junction in parallel, flux through the L-junction loop.
\newcommand{\fluxcircuit}[2]{%
  \draw (#2) ++(-1.6,0) coordinate (bl) ++(0,1.6) coordinate (tl)
        (#2) ++(-0.2,0) coordinate (bm) ++(0,1.6) coordinate (tm)
        (#2) ++(1.4,0) coordinate (br) ++(0,1.6) coordinate (tr);
  \draw (bl) to[#1, l=$C$] (tl) -- (tm) -- (tr) to[josephson, l=$E_J$] (br) -- (bm) -- (bl);
  \draw (tm) to[cute inductor, l_=$L$] (bm);
  \node[font=\small] at ($(#2)+(0.6,0.8)$) {$\Phi_{\rm ext}$};
}
